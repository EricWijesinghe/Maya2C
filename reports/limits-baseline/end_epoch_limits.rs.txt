//! Throwaway measurement harness (Lane 0, report only). Not for merge.
//! Times `Staking::end_epoch` on an in-memory state with 10,000 validators and
//! 100,000 delegations. Each repeat runs on a fresh clone of the same state.
#![allow(clippy::unwrap_used, clippy::expect_used, clippy::cast_precision_loss)]
use maya_staking::{Amount, Params, Participation, Staking};
use std::time::Instant;

const VALIDATORS: usize = 10_000;
const DELEGATIONS: usize = 100_000;
const REPEATS: usize = 5;

fn id_bytes(i: usize) -> [u8; 32] {
    let mut b = [0u8; 32];
    b[..8].copy_from_slice(&(i as u64).to_le_bytes());
    b
}

fn build(max_validators: u16) -> Staking {
    let mut params = Params::DEVNET;
    params.max_validators = max_validators;
    let mut s = Staking::new(params).unwrap();
    for i in 0..VALIDATORS {
        s.register(id_bytes(i), id_bytes(i), Amount::from_u64(1_000), 500).unwrap();
    }
    for d in 0..DELEGATIONS {
        let delegator = id_bytes(VALIDATORS + d);
        s.delegate(delegator, id_bytes(d % VALIDATORS), Amount::from_u64(1_000)).unwrap();
    }
    s
}

fn median(mut v: Vec<f64>) -> f64 {
    v.sort_by(|a, b| a.partial_cmp(b).unwrap());
    v[v.len() / 2]
}

fn main() {
    let rounds = 100u64;
    let mut part = Participation { rounds, ..Participation::default() };
    for i in 0..VALIDATORS {
        part.expected.insert(id_bytes(i), rounds);
        part.authored.insert(id_bytes(i), rounds);
    }
    for max_validators in [u16::try_from(VALIDATORS).unwrap(), 100u16] {
        let base = build(max_validators);
        let mut ms = Vec::new();
        let mut active = 0;
        for _ in 0..REPEATS {
            let mut s = base.clone();
            let t = Instant::now();
            let out = s.end_epoch(&part, Amount::from_u64(1_000_000_000)).unwrap();
            ms.push(t.elapsed().as_secs_f64() * 1e3);
            active = out.active.len();
        }
        println!(
            "end_epoch validators={VALIDATORS} delegations={DELEGATIONS} max_validators={max_validators} active_after={active} median_ms={:.3} runs_ms={:?}",
            median(ms.clone()),
            ms.iter().map(|x| (x * 1000.0).round() / 1000.0).collect::<Vec<_>>()
        );
    }
}
